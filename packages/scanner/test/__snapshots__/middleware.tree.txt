# Middleware

proxy.ts  proxy  export proxy
matcher: /dashboard/:path*, /shop/:path*, /old-home, /login (has cookie session)
runs on:
  /dashboard  page
  /dashboard/settings  page
  /login  page  [conditional]
rewrite /shop/* to /products -> /products
